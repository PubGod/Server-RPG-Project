# แผน: Minecraft Bedrock Race & Class Addon

## ภาพรวม

สร้าง addon สำหรับ Minecraft Bedrock Edition (มือถือ) ที่เพิ่มระบบเผ่า (Race) และอาชีพ (Class) โดยใช้ **Script API** (`@minecraft/server` + `@minecraft/server-ui`)

- เมื่อเข้าเกมครั้งแรก → แสดง UI เลือกเผ่า
- หลังเลือกเผ่า → แสดง UI เลือกอาชีพ
- ทุก ability เป็น **passive** (effect ต่อเนื่อง)
- เปลี่ยนเผ่า/อาชีพได้ด้วย chat command `!race` และ `!class`
- รองรับ Minecraft Bedrock **1.21+**

## manifest.json (Behavior Pack)

```json
{
  "format_version": 2,
  "header": {
    "name": "Race & Class System",
    "description": "เผ่าและอาชีพ",
    "uuid": "<UUID-1>",
    "version": [1, 0, 0],
    "min_engine_version": [1, 21, 0]
  },
  "modules": [
    {
      "type": "script",
      "language": "javascript",
      "uuid": "<UUID-2>",
      "version": [1, 0, 0],
      "entry": "scripts/main.js"
    }
  ],
  "dependencies": [
    { "module_name": "@minecraft/server", "version": "1.14.0" },
    { "module_name": "@minecraft/server-ui", "version": "1.3.0" }
  ]
}
```

Resource Pack manifest.json จะมีรูปแบบเดียวกันแต่ `type: "resources"` และไม่มี scripts/dependencies

## ระบบ Tag ของผู้เล่น

| ประเภท | Tag ที่ใช้ |
|---|---|
| เผ่า | `race:human`, `race:demon`, `race:elf`, `race:dwarf`, `race:dragon`, `race:angel` |
| อาชีพ | `class:farmer`, `class:warrior`, `class:hunter`, `class:blacksmith`, `class:archer` |

ใช้ `player.addTag()` / `player.hasTag()` / `player.removeTag()`

## main.js — Logic หลัก

```js
import { world, system } from '@minecraft/server';
import { showRaceUI } from './ui/raceUI.js';
import { applyRaceEffects } from './races/index.js';
import { applyClassBuffs } from './classes/index.js';

// 1. ตรวจผู้เล่นใหม่ → แสดง UI เลือกเผ่า
world.afterEvents.playerSpawn.subscribe(({ player, initialSpawn }) => {
  if (initialSpawn && !player.hasTag('race_selected')) {
    // delay 20 ticks ให้โลกโหลดก่อน
    system.runTimeout(() => showRaceUI(player), 20);
  }
});

// 2. Loop ทุก 100 ticks ใช้ passive effects
system.runInterval(() => {
  for (const player of world.getPlayers()) {
    applyRaceEffects(player);
    applyClassBuffs(player);
  }
}, 100);

// 3. Chat command สำหรับ reset
world.afterEvents.chatSend.subscribe(({ sender, message }) => {
  if (message === '!race') {
    sender.removeTag('race_selected');
    // ลบ race tags เก่า
    showRaceUI(sender);
  }
  if (message === '!class') {
    sender.removeTag('class_selected');
    showClassUI(sender);
  }
});
```

## raceUI.js — หน้า UI เลือกเผ่า

```js
import { ActionFormData } from '@minecraft/server-ui';
import { showClassUI } from './classUI.js';

const RACES = [
  { id: 'human',  name: '⚔ Human (มนุษย์)',  desc: 'สมดุล ฟื้นฟูเร็ว' },
  { id: 'demon',  name: '🔥 Demon (ปีศาจ)',   desc: 'แข็งแกร่งกลางคืน ทนไฟ' },
  { id: 'elf',    name: '🌿 Elf (เอลฟ์)',     desc: 'ว่องไว มองเห็นในที่มืด' },
  { id: 'dwarf',  name: '⛏ Dwarf (คนแคระ)', desc: 'แข็งแกร่ง ขุดเร็ว' },
  { id: 'dragon', name: '🐉 Dragon (มังกร)',  desc: 'ทนทาน มีพลังชีวิตสำรอง' },
  { id: 'angel',  name: '✨ Angel (ทูตสวรรค์)', desc: 'ลอยได้ ฟื้นตัวต่อเนื่อง' },
];

export function showRaceUI(player) {
  const form = new ActionFormData()
    .title('§l§6เลือกเผ่าของคุณ')
    .body('§7แต่ละเผ่ามีความสามารถพิเศษที่แตกต่างกัน\nเลือกได้เพียงครั้งเดียว (รีเซ็ตด้วย §e!race§7)');
  
  for (const race of RACES) {
    form.button(`§l${race.name}\n§7${race.desc}`);
  }
  
  form.show(player).then(response => {
    if (response.canceled) return;
    const selected = RACES[response.selection];
    // ลบ race tag เก่า
    for (const r of RACES) player.removeTag(`race:${r.id}`);
    player.addTag(`race:${selected.id}`);
    player.addTag('race_selected');
    player.sendMessage(`§aคุณเลือกเผ่า: §l${selected.name}`);
    // เปิด class UI ต่อ
    system.runTimeout(() => showClassUI(player), 20);
  });
}
```

    player.addTag(`class:${selected.id}`);

## Races — Passive Effects

### races/index.js
```js
import { applyHuman }  from './human.js';
import { applyDemon }  from './demon.js';
import { applyElf }    from './elf.js';
import { applyDwarf }  from './dwarf.js';
import { applyDragon } from './dragon.js';
import { applyAngel }  from './angel.js';

const RACE_MAP = {
  'race:human':  applyHuman,
  'race:demon':  applyDemon,
  'race:elf':    applyElf,
  'race:dwarf':  applyDwarf,
  'race:dragon': applyDragon,
  'race:angel':  applyAngel,
};

export function applyRaceEffects(player) {
  for (const [tag, fn] of Object.entries(RACE_MAP)) {
    if (player.hasTag(tag)) { fn(player); return; }
  }
}
```

## Effect API Reference (Bedrock Script API)

```js
// รูปแบบการใช้ addEffect
player.addEffect('speed', 120, { amplifier: 0, showParticles: false });
player.addEffect('strength', 120, { amplifier: 0, showParticles: false });
player.addEffect('resistance', 120, { amplifier: 0, showParticles: false });
player.addEffect('haste', 120, { amplifier: 0, showParticles: false });
player.addEffect('fire_resistance', 120, { showParticles: false });
player.addEffect('night_vision', 120, { showParticles: false });
player.addEffect('slow_falling', 120, { showParticles: false });
player.addEffect('regeneration', 120, { amplifier: 0, showParticles: false });
player.addEffect('absorption', 120, { amplifier: 1, showParticles: false });
player.addEffect('saturation', 120, { amplifier: 0, showParticles: false });
```

> ใช้ duration > runInterval ทุกครั้ง (100 ticks) เพื่อให้ effect ต่อเนื่อง

## การใช้งาน addon

1. วางโฟลเดอร์ `behavior_pack/` และ `resource_pack/` ใน:
   - Android: `Android/data/com.mojang.minecraftpe/files/games/com.mojang/`
   - iOS: `On My iPhone/Minecraft/games/com.mojang/`
2. เปิดเกม → Settings → Storage → Global Resources/Behavior
3. เริ่มโลกใหม่ → เปิด pack ทั้งคู่
4. เข้าเกม → หน้าต่าง UI จะปรากฏ
